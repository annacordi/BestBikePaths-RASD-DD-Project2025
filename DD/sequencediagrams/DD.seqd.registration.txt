@startuml
title DD - User Registration Runtime View

skinparam style strictuml

actor User
participant "Registration Page" as APP
participant "Registration Handler" as RH
participant "Permission Manager" as PM
participant "Database Query Manager" as QM
participant "Notification Manager" as NM

User -> APP: Enter registration data
activate APP

APP -> RH: Validate input data
activate RH

alt Invalid data
    RH --> APP: Return validation errors
    APP --> User: Show error messages
else Valid data
    RH --> APP: Data valid
    
    RH -> QM: Check if email exists
    activate QM
    
    alt Email already exists
        QM --> RH: Email found
        deactivate QM
        RH --> APP: Email already registered
        deactivate RH
        APP --> User: Show error message
    else Email not exists
        QM --> RH: Email available
        
        RH -> QM: Save user credentials
        activate QM
        QM --> RH: User saved
        deactivate QM
        
        RH --> APP: User created successfully
        deactivate RH
        
        APP --> User: Show permissions\nconfiguration screen
        
        User -> APP: Select permission preferences\n(GPS, Auto-detection, Sensors)
        
        APP -> PM: Configure sensor permissions
        activate PM
        
        PM -> QM: Save permission preferences
        activate QM
        
        alt Permission update fails
            QM --> PM: Update failed
            deactivate QM
            PM --> APP: Permission update error
            deactivate PM
            APP --> User: Show warning\n(Can configure later)
        else Permission update succeeds
            QM --> PM: Permissions saved
            deactivate QM
            PM --> APP: Permissions updated
            deactivate PM
        end
        
        APP -> NM: Request verification email
        activate NM
        
        NM -> NM: Send verification email
        NM --> APP: Email sent
        deactivate NM
        
        APP --> User: Show success message\n(Check email)
        deactivate APP
    end
end

@enduml
