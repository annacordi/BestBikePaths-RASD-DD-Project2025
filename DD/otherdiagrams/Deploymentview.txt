' Mobile Device
node "Mobile Device" <<device>> {
    component "BBP Mobile App" <<application>> as MobApp
    component "Maps" <<service>> as MapService
    component "Weather API" <<service>> as MobWeather
    component "Edge Firewall" <<security>> as MobFirewall
}

' Backend Deployment
package "BBP Backend Deployment Perimeter" {
  
  ' Cluster Node 1
  node "Cluster Node 1" <<server>> as S1{

      node "Docker Container" <<execution environment>> as Docker1 {
        database "Load Balancer\nNGINX" <<load balancer>> as LB
      }
    }
  }
  
  ' Cluster Node 2

      }
    }
  }
  
  ' Cluster Node 4
  node "Cluster Node 4" <<server>> as S4 {
    node "Ubuntu 24 LTS" <<operating system>> as OS4 {
      node "Docker Container" <<execution environment>> as Docker4 {
        component "Email Service\nSendMail/SMTP" <<service>> as EmailService
      }
    }
  }
  
  ' Cluster Node 3
